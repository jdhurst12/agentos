Quickstart

import { agent } from '@framers/agentos';

const strategist = agent({
  provider: 'anthropic',
  instructions: 'Operate with precision.',
  memory: { types: ['episodic', 'semantic'] },
  personality: { openness: 0.91, conscientiousness: 0.94 },
});

const session = strategist.session('founder');
await session.send('Map the market and find leverage.');
